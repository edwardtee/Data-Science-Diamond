knn_regressor = KNeighborsRegressor(n_neighbors=25, weights='distance')
knn_regressor.fit(X_train, y_train)
y_pred = knn_regressor.predict(X_test)

# R² Score: Tells you the percentage of variance in diamond prices your model explains (1.0 is perfect)
r2 = r2_score(y_test, y_pred)
print(f"R² Score: {r2:.4f} (Your model explains {r2 * 100:.2f}% of the price variance)")

# Mean Absolute Error: The average physical dollar amount your predictions are off by
mae = mean_absolute_error(y_test, y_pred)
print(f"Mean Absolute Error (MAE): ${mae:.2f}")

# Root Mean Squared Error: Similar to MAE, but penalizes larger misses more heavily
rmse = np.sqrt(mean_squared_error(y_test, y_pred))
print(f"Root Mean Squared Error (RMSE): ${rmse:.2f}")
